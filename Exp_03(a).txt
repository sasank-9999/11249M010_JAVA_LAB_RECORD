import java.util.Scanner;
class EmployeeString {
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter frist employee name:");
        String emp1 = sc.nextLine();
        System.out.println("Enter second employee name:");
        String emp2 = sc.nextLine();

        if(emp1.equalsIgnoreCase(emp2))
            System.out.println("Both are in the same department.");
        else
            System.out.println("They are not in same department.");
        sc.close();

    }
    
}
