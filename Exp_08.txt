    void studentDetails();
}
interface Marks{
    void marksDetails();
}
class Result implements Interface, Marks{

        System.out.println("Name: Lohitha");
        System.out.println("Roll No: 384");

    }
    public void marksDetails(){

    } 
}

        Result r = new Result();
        r.studentDetails();
        r.marksDetails();
    }
}
