class Sales {
    public static void main(String args[]){
        int[] sales = {1200,1500,1800,900,2500,3000,1700,2200,1600,1400,2800,2000};
        int max = sales[0];
        int min = sales[0];
        for (int i = 1; i< sales.length; i++){
            if(sales[i] > max)
                max = sales[i];
            if (sales[i] < min)
                min = sales[i];
        }
        System.out.println("Maximum sales = "+ max);
        System.out.println("Minimum sales = "+min);

    }
}
